```python
import pandas as pd
import hashlib

# โหลดข้อมูลจริง
df = pd.read_csv('original_student_data.csv')

# สร้าง Anonymous ID
def anonymize_id(real_id):
    # ใช้ SHA256 hash + salt
    salt = "your-secret-salt-here"  # เปลี่ยนเป็น secret ของคุณ
    hashed = hashlib.sha256(f"{real_id}{salt}".encode()).hexdigest()
    return f"S{hashed[:8]}"  # ใช้ 8 ตัวอักษรแรก

df['StudentID'] = df['RealStudentID'].apply(anonymize_id)

# ลบคอลัมน์ระบุตัวตน
df = df.drop(columns=['RealStudentID', 'Name', 'Email', 'PhoneNumber'])

# บันทึก
df.to_csv('dataset/ID_Group.csv', index=False)
```

ขั้นตอนที่ 4: จัดรูปแบบให้ตรง
------------------------------

ใช้สคริปต์นี้:

```python
# format_real_data.py
import pandas as pd
import numpy as np

def format_attendance_data(input_file, output_file):
    """จัดรูปแบบข้อมูลการเข้าเรียน"""
    df = pd.read_csv(input_file)

    # ตรวจสอบว่ามีคอลัมน์ครบ
    required_cols = ['StudentID'] + [f'Week{i}' for i in range(1, 16)]

    for col in required_cols:
        if col not in df.columns:
            print(f"⚠️  Missing column: {col}")
            return False

    # แปลงเป็น 0/1
    week_cols = [f'Week{i}' for i in range(1, 16)]
    for col in week_cols:
        df[col] = df[col].apply(lambda x: 1 if x in [1, 'Y', 'Yes', 'P', 'Present'] else 0)

    # บันทึก
    df[required_cols].to_csv(output_file, index=False)
    print(f"✓ Saved: {output_file}")
    return True

def format_score_data(input_file, output_file, score_columns):
    """จัดรูปแบบข้อมูลคะแนน"""
    df = pd.read_csv(input_file)

    required_cols = ['StudentID'] + score_columns

    # ตรวจสอบคอลัมน์
    for col in required_cols:
        if col not in df.columns:
            print(f"⚠️  Missing column: {col}")
            return False

    # แปลงเป็น numeric
    for col in score_columns:
        df[col] = pd.to_numeric(df[col], errors='coerce')

    # บันทึก
    df[required_cols].to_csv(output_file, index=False)
    print(f"✓ Saved: {output_file}")
    return True

# ใช้งาน:
format_attendance_data('raw/sj111_attendance.csv',
                      'dataset/SJ111777_Attendance15weeks.csv')

format_score_data('raw/sj111_score.csv',
                 'dataset/SJ111777_Score.csv',
                 ['Tasks(100Points)', 'Quiz(150Points)', 'Project(100Points)',
                  'Midterm(150Points)', 'Final(150Points)', 'ScoreGrade(100Points)'])
```

6. ข้อควรระวังและ Best Practices
================================================================================

⚠️  ข้อควรระวัง
----------------

1. Data Leakage
   ✗ อย่าใช้ข้อมูลอนาคตทำนายอดีต
   ✗ อย่าใช้ข้อมูล test ในการ train
   ✗ แยก validation set ชัดเจน

2. Overfitting
   ✗ R² = 1.00 หรือใกล้เคียง = น่าสงสัย
   ✗ Training R² >> Test R² = overfit
   ✓ ใช้ Cross-validation
   ✓ ใช้ Regularization (Lasso, Ridge)

3. Class Imbalance
   ✗ ถ้านักศึกษาส่วนใหญ่ได้เกรดดี
   ✓ ใช้ stratified split
   ✓ พิจารณา SMOTE หรือ class weights

4. Feature Selection
   ✗ อย่าใช้ features ที่ไม่เกี่ยวข้อง
   ✓ ตรวจสอบ feature importance
   ✓ ลบ features ที่ correlation ต่ำ

5. Model Interpretability
   ✗ อย่าใช้ black box models เพียงอย่างเดียว
   ✓ อธิบายได้ว่าทำไมโมเดลทำนายแบบนั้น
   ✓ ใช้ SHAP หรือ LIME สำหรับ explanation

-----------------

7. การประเมินและ Validation
================================================================================

Validation Strategy
-------------------

1. Train/Test Split (80/20)
2. Cross-Validation (5-fold)
3. Holdout Set (ข้อมูลเทอมล่าสุด)

```python
from sklearn.model_selection import cross_val_score, KFold

# K-Fold Cross-Validation
kfold = KFold(n_splits=5, shuffle=True, random_state=42)
cv_scores = cross_val_score(model, X, y, cv=kfold, scoring='r2')

print(f"CV R² Scores: {cv_scores}")
print(f"CV R² Mean: {cv_scores.mean():.4f} (+/- {cv_scores.std():.4f})")
```

Metrics ที่ต้องดู
------------------

```python
from sklearn.metrics import r2_score, mean_absolute_error, mean_squared_error

# R² Score (สูงกว่า 0.80 = ดี)
r2 = r2_score(y_test, y_pred)

# MAE (ต่ำกว่า 5 คะแนน = ดี)
mae = mean_absolute_error(y_test, y_pred)

# RMSE (ต่ำกว่า 7 คะแนน = ดี)
rmse = np.sqrt(mean_squared_error(y_test, y_pred))

# MAPE (ต่ำกว่า 10% = ดี)
mape = np.mean(np.abs((y_test - y_pred) / y_test)) * 100

print(f"R²: {r2:.4f}")
print(f"MAE: {mae:.2f} points")
print(f"RMSE: {rmse:.2f} points")
print(f"MAPE: {mape:.2f}%")
```

เกณฑ์ประเมินโมเดล
-------------------

| Metric | Excellent | Good | Fair | Poor |
|--------|-----------|------|------|------|
| R² | > 0.90 | 0.80-0.90 | 0.70-0.80 | < 0.70 |
| MAE | < 3 | 3-5 | 5-7 | > 7 |
| RMSE | < 4 | 4-7 | 7-10 | > 10 |
| MAPE | < 5% | 5-10% | 10-15% | > 15% |

สร้างไฟล์ใหม่: prepare_real_data.py
------------------------------------

```python
"""
Prepare Real Student Data for ML Pipeline
==========================================
สคริปต์นี้เตรียมข้อมูลจริงให้พร้อมใช้งาน
"""

import pandas as pd
import hashlib
from pathlib import Path

class RealDataPreparator:
    """เตรียมข้อมูลจริงให้พร้อมใช้"""

    def __init__(self, raw_dir='raw_data', output_dir='dataset'):
        self.raw_dir = Path(raw_dir)
        self.output_dir = Path(output_dir)
        self.output_dir.mkdir(exist_ok=True)

        # Secret salt สำหรับ hashing (เปลี่ยนเป็นของคุณ)
        self.SALT = "change-this-to-your-secret-salt"

    def anonymize_student_id(self, real_id):
        """สร้าง Anonymous ID จาก Real ID"""
        hashed = hashlib.sha256(f"{real_id}{self.SALT}".encode()).hexdigest()
        return f"S{hashed[:8].upper()}"

    def prepare_id_group(self, input_file):
        """เตรียมไฟล์ ID_Group"""
        print("📋 Preparing ID_Group.csv...")

        df = pd.read_csv(self.raw_dir / input_file)

        # Anonymize
        df['StudentID'] = df['RealStudentID'].apply(self.anonymize_student_id)

        # เก็บเฉพาะคอลัมน์ที่ต้องการ
        df = df[['StudentID', 'Group', 'GPA']]

        # Validate
        assert df['GPA'].between(0, 4).all(), "GPA out of range!"
        assert df['StudentID'].is_unique, "Duplicate StudentIDs!"

        # Save
        output_file = self.output_dir / 'ID_Group.csv'
        df.to_csv(output_file, index=False)
        print(f"✓ Saved: {output_file}")
        print(f"  Students: {len(df)}")

    def prepare_attendance(self, input_file, output_filename):
        """เตรียมไฟล์การเข้าเรียน"""
        print(f"📅 Preparing {output_filename}...")

        df = pd.read_csv(self.raw_dir / input_file)

        # Anonymize
        df['StudentID'] = df['RealStudentID'].apply(self.anonymize_student_id)

        # แปลงเป็น 0/1
        week_cols = [f'Week{i}' for i in range(1, 16)]
        for col in week_cols:
            df[col] = df[col].apply(lambda x: 1 if x in [1, 'Y', 'Yes', 'P', 'Present', True] else 0)

        # เก็บเฉพาะคอลัมน์ที่ต้องการ
        df = df[['StudentID'] + week_cols]

        # Save
        output_file = self.output_dir / output_filename
        df.to_csv(output_file, index=False)
        print(f"✓ Saved: {output_file}")

    def prepare_scores(self, input_file, output_filename, score_cols):
        """เตรียมไฟล์คะแนน"""
        print(f"📊 Preparing {output_filename}...")

        df = pd.read_csv(self.raw_dir / input_file)

        # Anonymize
        df['StudentID'] = df['RealStudentID'].apply(self.anonymize_student_id)

        # แปลงเป็น numeric
        for col in score_cols:
            df[col] = pd.to_numeric(df[col], errors='coerce')

        # เก็บเฉพาะคอลัมน์ที่ต้องการ
        df = df[['StudentID'] + score_cols]

        # Validate
        for col in score_cols:
            max_score = int(col.split('(')[1].split('Points')[0])
            assert df[col].between(0, max_score, na=True).all(), f"{col} out of range!"

        # Save
        output_file = self.output_dir / output_filename
        df.to_csv(output_file, index=False)
        print(f"✓ Saved: {output_file}")

    def prepare_all(self):
        """เตรียมทุกไฟล์"""
        print("=" * 60)
        print("🚀 Preparing Real Data for ML Pipeline")
        print("=" * 60)

        # ID & Group
        self.prepare_id_group('students.csv')

        # Course 1
        self.prepare_attendance('sj111_attendance.csv', 'SJ111777_Attendance15weeks.csv')
        self.prepare_scores('sj111_scores.csv', 'SJ111777_Score.csv',
                           ['Tasks(100Points)', 'Quiz(150Points)', 'Project(100Points)',
                            'Midterm(150Points)', 'Final(150Points)', 'ScoreGrade(100Points)'])

        # Course 2
        self.prepare_attendance('sj220_attendance.csv', 'SJ220777_Attendance15weeks.csv')
        self.prepare_scores('sj220_scores.csv', 'SJ220777_Score.csv',
                           ['Tasks(100Points)', 'Quiz(90Points)', 'Project(90Points)',
                            'Midterm(180Points)', 'Final(180Points)', 'ScoreGrade(100Points)'])

        # Course 3
        self.prepare_attendance('sj333_attendance.csv', 'SJ333777_Attendance15weeks.csv')
        self.prepare_scores('s333_scores.csv', 'S333777_Score.csv',
                           ['Tasks(100Points)', 'Project(100Points)',
                            'Midterm(100Points)', 'Final(100Points)', 'ScoreGrade(100Points)'])

        print("\n" + "=" * 60)
        print("✅ All files prepared successfully!")
        print("=" * 60)
        print("\nNext steps:")
        print("1. Verify files in dataset/ folder")
        print("2. Update code to use 'dataset' directory")
        print("3. Run: python3 student_grade_prediction.py")


# ใช้งาน:
if __name__ == '__main__':
    preparator = RealDataPreparator()
    preparator.prepare_all()
```
